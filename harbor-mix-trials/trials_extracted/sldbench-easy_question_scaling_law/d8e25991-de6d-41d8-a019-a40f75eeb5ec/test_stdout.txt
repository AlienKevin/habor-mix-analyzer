# Coefficients for: loss = a * log10(flops)^2 + b * log10(flops) + c
# where flops = num_params * parallel_size
COEFFS = {
    "mmlu": [0.01147626, -0.06297043, -0.48036465],
    "parsinlu_qa_mc": [-0.05656740, 0.09890584, -0.43495072],

    "abstract_narrative_understanding": [-0.00100210, 0.18472699, -0.54314071],
}

def law(input_data: list[dict[str, float]], group: str) -> list[dict[str, float]]:
    """
    Predicts output variables based on input variables according to a discovered scaling law.
    
    Formula: loss = a * log10(num_params * parallel_size)^2 + b * log10(num_params * parallel_size) + c
    """
    a, b, c = COEFFS[group]
    results = []
    for point in input_data:
        num_params = point.get("num_params", point.get("flops", 1.0))
        parallel_size = point.get("parallel_size", 1.0)
        flops = num_params * parallel_size
        log_flops = np.log10(flops)
        loss = a * log_flops**2 + b * log_flops + c
        results.append({"loss": float(loss)})
    return results
