import numpy as np
from typing import List, Dict

# Fitted parameters for each group
# Model: brier_score = a + b*log10(num_params) + c*log10(parallel_size)
# The predicted loss is the brier_score value.

GROUP_PARAMS = {
    "mmlu": {"a": -0.490470, "b": -0.065802, "c": -0.004706},
    "parsinlu_qa_mc": {"a": -0.424733, "b": 0.048100, "c": -0.062579},
    "arithmetic": {"a": -0.252913, "b": 0.121192, "c": -0.039928},
    "hindu_knowledge": {"a": -0.416483, "b": -0.028796, "c": -0.053552},
    "analogical_similarity": {"a": -0.535680, "b": 0.021603, "c": -0.033528},
    "conceptual_combinations": {"a": -0.413147, "b": 0.071087, "c": -0.043467},
    "hellaswag": {"a": -0.061479, "b": 0.067109, "c": 0.003274},
    "arc": {"a": -0.099293, "b": 0.086818, "c": 0.008173},
    "abstract_narrative_understanding": {"a": -0.523892, "b": 0.221831, "c": 0.112867},
}

def law(input_data: List[Dict[str, float]], group: str) -> List[Dict[str, float]]:
    """
    Predicts output variables based on input variables according to a discovered scaling law.

    The scaling law relates brier_score (loss) to num_params and parallel_size:
    brier_score = a_group + b_group * log10(num_params) + c_group * log10(parallel_size)

    Args:
        input_data: A list of dictionaries, where each dictionary is a single data
                    point containing input variable names as keys and their
                    corresponding values.
        group: The name of the experimental group for which to make predictions.
                The functional form of the law must be the same for all groups,
                but the constant parameters/coefficients can differ per group.

    Returns:
        A list of dictionaries, corresponding to the input_data list, with each
        dictionary containing the predicted output variable(s).
    """
    if group not in GROUP_PARAMS:
        raise ValueError(f"Unknown group: {group}")

    params = GROUP_PARAMS[group]
    a = params["a"]
    b = params["b"]
    c = params["c"]

    results = []
    for dp in input_data:
        num_params = dp["num_params"]
        parallel_size = dp["parallel_size"]

        # Compute log10 values, handling edge cases
        log_N = np.log10(num_params) if num_params > 0 else 0.0
        log_P = np.log10(parallel_size) if parallel_size > 0 else 0.0

        # Predict brier_score (which is the loss)
        brier_score = a + b * log_N + c * log_P

        # Return as loss
        results.append({"loss": brier_score})

    return results
