"""
Optimized matrix square root solver.

Computes the principal matrix square root X such that X @ X = A,
where X is the unique matrix with eigenvalues having non-negative real parts.
"""

import logging
import numpy as np
import scipy.linalg
from typing import Any


class Solver:
    def solve(self, problem: dict, **kwargs) -> dict[str, dict[str, list[list[complex]]]]:
        """
        Solves the matrix square root problem using scipy.linalg.sqrtm.

        Computes the principal matrix square root X such that X @ X = A.

        :param problem: A dictionary representing the matrix square root problem.
            Expected keys: "matrix" - 2D array of complex numbers.
        :return: A dictionary with key "sqrtm" containing a dictionary with key:
                 "X": A list of lists representing the principal square root matrix X.
        """
        # Extract matrix from problem
        matrix_data = problem.get("matrix")

        if matrix_data is None:
            logging.error("Problem does not contain 'matrix'.")
            return {"sqrtm": {"X": []}}

        # Convert input to numpy array efficiently
        if isinstance(matrix_data, np.ndarray):
            A = matrix_data.astype(complex)
        else:
            A = np.array(matrix_data, dtype=complex)

        n = A.shape[0]

        # Handle empty matrix edge case
        if n == 0:
            return {"sqrtm": {"X": []}}

        # Compute the principal matrix square root
        # Using disp=False to suppress stdout warnings
        # Using blocksize=128 for better cache utilization on larger matrices
        try:
            result = scipy.linalg.sqrtm(A, disp=False, blocksize=128)
            # Handle 1x1 matrices which return just the array, not a tuple
            X = result[0] if isinstance(result, tuple) else result
        except Exception as e:
            logging.error(f"scipy.linalg.sqrtm failed: {e}")
            return {"sqrtm": {"X": []}}

        # Convert to list for JSON serialization
        return {"sqrtm": {"X": X.tolist()}}--- End Solver ---
