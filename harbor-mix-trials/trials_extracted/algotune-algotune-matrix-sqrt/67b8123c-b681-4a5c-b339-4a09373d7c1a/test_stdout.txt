import numpy as np
import logging
from typing import Any

logger = logging.getLogger(__name__)


class Solver:
    def __init__(self):
        # Pre-import scipy.linalg to avoid import overhead during solve
        import scipy.linalg
        self._sqrtm = scipy.linalg.sqrtm
    
    def solve(self, problem, **kwargs) -> Any:
        """
        Solves the matrix square root problem.
        
        Computes the principal matrix square root X such that X @ X = A.
        Optimized by calling scipy.linalg.sqrtm without deprecated disp argument.
        
        :param problem: A dictionary representing the matrix square root problem.
        :return: A dictionary with key "sqrtm" containing the solution.
        """
        A = problem["matrix"]
        
        # Convert input to numpy array if needed
        if not isinstance(A, np.ndarray):
            A = np.asarray(A, dtype=complex)
        elif A.dtype != complex:
            A = A.astype(complex)
        
        try:
            # Call sqrtm without disp=False to get just the matrix
            # This is faster than the deprecated disp=False which returns a tuple
            X = self._sqrtm(A)
        except Exception as e:
            logger.error(f"scipy.linalg.sqrtm failed: {e}")
            return {"sqrtm": {"X": []}}
        
        # Check for non-finite values in the result
        if not np.all(np.isfinite(X)):
            logger.warning("sqrtm returned non-finite values")
            return {"sqrtm": {"X": []}}
        
        return {"sqrtm": {"X": X.tolist()}}
