import numpy as np
import torch


class Solver:
    def solve(self, problem, **kwargs):
        A = problem["matrix"]

        # Handle 1x1 case directly
        if A.shape[0] == 1:
            return {"sqrtm": {"X": np.sqrt(A.astype(complex)).tolist()}}

        n = A.shape[0]

        if n < 50:
            # For small matrices, numpy eig is faster due to less overhead
            d, V = np.linalg.eig(A)
            sqrt_d = np.sqrt(d.astype(complex))
            V_sqrtD = V * sqrt_d[np.newaxis, :]
            X = V_sqrtD @ np.linalg.inv(V)
        else:
            # For larger matrices, torch is significantly faster
            A_t = torch.tensor(A, dtype=torch.complex128)
            d_t, V_t = torch.linalg.eig(A_t)
            sqrt_d_t = torch.sqrt(d_t)
            V_sqrtD_t = V_t * sqrt_d_t.unsqueeze(0)
            X_t = V_sqrtD_t @ torch.linalg.inv(V_t)
            X = X_t.numpy()

        return {"sqrtm": {"X": X.tolist()}}
--- End Solver ---
