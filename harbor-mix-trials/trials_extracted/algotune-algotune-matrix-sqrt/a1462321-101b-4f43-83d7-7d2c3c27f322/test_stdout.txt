from typing import Any
import numpy as np
from scipy.linalg._matfuncs_schur_sqrtm import recursive_schur_sqrtm


class Solver:
    def solve(self, problem, **kwargs) -> Any:
        A = problem["matrix"]
        a = np.asarray(A)

        # Coerce to a supported dtype: 'fdFD' (float32/64, complex64/128).
        kind = a.dtype.kind
        if kind in ('U', 'S', 'O'):
            # Strings or Python objects: parse element-wise to complex.
            a = np.array(A, dtype=np.complex128)
        elif not np.issubdtype(a.dtype, np.inexact):
            a = a.astype(np.float64)
        elif a.dtype == np.float16:
            a = a.astype(np.float32)
        elif a.dtype.char == 'G':
            a = a.astype(np.complex128)
        elif a.dtype.char == 'g':
            a = a.astype(np.float64)

        if not a.flags['C_CONTIGUOUS']:
            a = np.ascontiguousarray(a)

        try:
            X, _ill, _sing, info = recursive_schur_sqrtm(a)
            if info < 0:
                return {"sqrtm": {"X": []}}
        except Exception:
            return {"sqrtm": {"X": []}}

        return {"sqrtm": {"X": X.tolist()}}
--- End Solver ---
