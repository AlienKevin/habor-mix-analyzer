import numpy as np
import scipy.linalg
from threadpoolctl import threadpool_limits

class Solver:
    def __init__(self):
        # Initialize threadpool limits to 1 to avoid OpenBLAS thread contention
        self.thread_limit = threadpool_limits(limits=1)
        self.thread_limit.__enter__()

    def __del__(self):
        if hasattr(self, 'thread_limit'):
            try:
                self.thread_limit.__exit__(None, None, None)
            except Exception:
                pass

    def solve(self, problem: dict, **kwargs) -> dict:
        try:
            # Ensure the input is a complex numpy array
            A = np.asarray(problem["matrix"], dtype=complex)
            # disp=False prevents printing to stdout/stderr
            X, _ = scipy.linalg.sqrtm(A, disp=False)
            return {"sqrtm": {"X": X.tolist()}}
        except Exception as e:
            return {"sqrtm": {"X": []}}
