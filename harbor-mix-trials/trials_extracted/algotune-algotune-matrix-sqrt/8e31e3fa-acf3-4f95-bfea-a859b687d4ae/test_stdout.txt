import numpy as np
from scipy.linalg import schur, solve_triangular
from scipy.linalg._matfuncs_sqrtm import _sqrtm_triu

class Solver:
    def solve(self, problem, **kwargs):
        A = problem["matrix"]
        A = np.asarray(A, dtype=complex)
        try:
            T, Z = schur(A, output='complex')
            R = _sqrtm_triu(T, blocksize=64)
            X = Z @ R @ Z.conj().T
        except Exception:
            return {"sqrtm": {"X": []}}
        return {"sqrtm": {"X": X.tolist()}}
