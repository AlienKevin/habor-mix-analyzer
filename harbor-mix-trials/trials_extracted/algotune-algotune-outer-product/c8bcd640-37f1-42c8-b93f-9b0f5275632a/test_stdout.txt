import numpy as np


class Solver:
    def solve(self, problem, **kwargs) -> np.ndarray:
        vec1, vec2 = problem
        # Ensure numpy arrays (handles list inputs), then use broadcasting.
        # v1[:, None] * v2 is faster than np.outer by avoiding its
        # internal ravel/asarray overhead and extra Python-level checks.
        a = np.asarray(vec1)
        b = np.asarray(vec2)
        return a[:, None] * b
