import numpy as np
from typing import Any

class Solver:
    def solve(self, problem, **kwargs) -> Any:
        vec1, vec2 = problem
        n = len(vec1)
        # einsum is faster for smaller vectors due to less overhead
        # For larger vectors, use broadcasting (similar performance to np.outer)
        if n < 3000:
            return np.einsum('i,j->ij', vec1, vec2)
        else:
            return vec1[:, np.newaxis] * vec2[np.newaxis, :]
