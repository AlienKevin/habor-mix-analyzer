import numpy as np
import numba as nb

@nb.njit
def _outer_numba(vec1, vec2):
    n = len(vec1)
    m = len(vec2)
    result = np.empty((n, m), dtype=np.float64)
    for i in range(n):
        for j in range(m):
            result[i, j] = vec1[i] * vec2[j]
    return result

# Pre-compile at module load time
_outer_numba(np.array([1.0, 2.0]), np.array([3.0, 4.0]))

        n = len(vec1)

            return np.outer(vec1, vec2)
