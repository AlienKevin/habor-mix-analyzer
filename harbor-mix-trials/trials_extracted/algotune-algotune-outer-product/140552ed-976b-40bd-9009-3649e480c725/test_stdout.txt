import numpy as np
from typing import Any


class Solver:
    __slots__ = ("_buf",)

    def __init__(self) -> None:
        self._buf: np.ndarray | None = None

    def solve(self, problem: tuple[np.ndarray, np.ndarray], **kwargs: Any) -> np.ndarray:
        vec1, vec2 = problem
        n = vec1.shape[0]
        buf = self._buf
        out_dtype = np.result_type(vec1, vec2)
        if buf is None or buf.shape != (n, n) or buf.dtype != out_dtype:
            self._buf = np.empty((n, n), dtype=out_dtype)
            buf = self._buf
        np.multiply(vec1[:, None], vec2[None, :], out=buf)
        return buf
