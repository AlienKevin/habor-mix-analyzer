import numpy as np


class Solver:
    def solve(self, problem, **kwargs) -> Any:
        vec1, vec2 = problem
        n = vec1.shape[0]

        if n == 0:
            return np.empty((0, vec2.shape[0]), dtype=np.result_type(vec1, vec2))
        if n == 1:
            return np.multiply(vec1.reshape(1, 1), vec2.reshape(1, -1))

        return np.multiply(vec1[:, None], vec2)
