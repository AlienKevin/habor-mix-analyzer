import numpy as np
import numba

@numba.njit(fastmath=True)
def _outer_numba(v1, v2):
    """Numba-accelerated outer product for small arrays."""
    n = v1.shape[0]
    out = np.empty((n, n), dtype=np.float64)
    for i in range(n):
        vi = v1[i]
        for j in range(n):
            out[i, j] = vi * v2[j]
    return out

class Solver:
    """Optimized solver for vector outer product.
    
    Uses numba-accelerated manual loop for small arrays (n <= 800),
    and numpy's optimized outer product for larger arrays.
    """
    THRESHOLD = 800
    
    def __init__(self):
        # Pre-compile the numba function (compilation time not counted toward solve())
        dummy = np.zeros(2, dtype=np.float64)
        _outer_numba(dummy, dummy)
    
    def solve(self, problem, **kwargs):
        vec1, vec2 = problem
        n = len(vec1)
        if n <= self.THRESHOLD:
            # For small arrays, use numba for speed
            return _outer_numba(
                np.asarray(vec1, dtype=np.float64),
                np.asarray(vec2, dtype=np.float64)
            )
        # For larger arrays, numpy's outer is optimal
        return np.outer(vec1, vec2)
