import numpy as np
from typing import Any


class Solver:
    def solve(self, problem, **kwargs) -> Any:
        vec1, vec2 = problem
        a = np.asarray(vec1)
        b = np.asarray(vec2)
        # Use einsum for small/medium arrays (faster up to ~2500 elements),
        # fall back to np.outer for very large arrays
        if len(a) <= 2500:
            return np.einsum('i,j->ij', a, b)
        else:
            return np.outer(a, b)
