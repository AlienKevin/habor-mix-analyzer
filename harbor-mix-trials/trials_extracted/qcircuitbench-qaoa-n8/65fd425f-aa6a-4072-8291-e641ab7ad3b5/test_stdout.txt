from qiskit import transpile
from scipy.optimize import minimize

def run_and_analyze(circuit, aer_sim, graph):
    def objective(params):
        param_dict = {}
        for i, param in enumerate(circuit.parameters):
            param_dict[param] = [params[i]]
        parameter_binds = [param_dict]
        transpiled_circuit = transpile(circuit, aer_sim)
        job = aer_sim.run(transpiled_circuit, parameter_binds=parameter_binds, shots=1024)
        result = job.result()
        counts = result.get_counts()
        expectation = compute_expectation(counts, graph)
        return -expectation
    
    initial_params = [0.1] * circuit.num_parameters
    result = minimize(objective, initial_params, method='COBYLA')
    
    optimal_params = result.x
    param_dict = {}
    for i, param in enumerate(circuit.parameters):
        param_dict[param] = [optimal_params[i]]
    parameter_binds = [param_dict]
    transpiled_circuit = transpile(circuit, aer_sim)
    job = aer_sim.run(transpiled_circuit, parameter_binds=parameter_binds, shots=4096)
    result = job.result()
    counts = result.get_counts()
    
    sorted_counts = sorted(counts.items(), key=lambda x: x[1], reverse=True)
    top_two = [sorted_counts[0][0], sorted_counts[1][0]]
    return top_two
