def run_and_analyze(circuit, aer_sim):

    # The input register has 5 qubits encoding 3 trits
    # 2^5 = 32 possible measurements, but only 27 should be valid
    # (corresponding to 3^3 = 27 possible trit triples)
    
    # Decode: interpret 5-bit number as base-3 trit triple
    # Number 0-26 maps to (t0, t1, t2) where num = t0 + 3*t1 + 9*t2
    
    def bits_to_trits(bits):
        num = int(bits, 2)
        if num >= 27:
            return None
        t2 = num % 3; num //= 3
        t1 = num % 3; num //= 3
        t0 = num % 3
        return (t0, t1, t2)
    
    # Collect y vectors (excluding zero)
    y_counts = {}
    for bits, count in input_counts.items():
        y = bits_to_trits(bits)
        if y is not None and y != (0, 0, 0):
            y_counts[y] = y_counts.get(y, 0) + count
