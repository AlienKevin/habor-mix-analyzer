code_string = '''from qiskit import transpile

def run_and_analyze(circuit, aer_sim):
    def clean_key(key):
        return key.replace(" ", "")

    def bits_to_int(bits, little_endian):
        if little_endian:
            total = 0
            for index, bit in enumerate(bits):
                if bit == "1":
                    total += 1 << index
            return total
        return int(bits, 2)

    def int_to_vec(value, little_trits):
        digits = []
        current = value
        for _ in range(3):
            digits.append(current % 3)
            current //= 3
        if little_trits:
            return digits
        return [digits[2], digits[1], digits[0]]

    def vec_to_int(vector, little_trits):
        if little_trits:
            return vector[0] + 3 * vector[1] + 9 * vector[2]
        return 9 * vector[0] + 3 * vector[1] + vector[2]

    def add_vec(value, vector, scale, little_trits):
        base = int_to_vec(value, little_trits)
        added = [(base[index] + scale * vector[index]) % 3 for index in range(3)]
        return vec_to_int(added, little_trits)

    def normalize(vector):
        if vector == [0, 0, 0]:
            return vector
        first = next(item for item in vector if item != 0)
        if first == 2:
            return [(2 * item) % 3 for item in vector]
        return vector

    compiled = transpile(circuit, aer_sim)
    result = aer_sim.run(compiled, shots=20000, seed_simulator=314159).result()
    counts = result.get_counts()

    observations = []
    for key, count in counts.items():
        key = clean_key(key)
        if len(key) < 10:
            key = key.zfill(10)
        classical_low_to_high = key[::-1]
        input_bits = classical_low_to_high[:5]
        output_bits = classical_low_to_high[5:10]
        observations.append((input_bits, output_bits, count))

    best_score = -1
    best_vector = [1, 0, 0]

    for little_bits in (True, False):
        input_to_output_counts = {}
        for input_bits, output_bits, count in observations:
            value = bits_to_int(input_bits, little_bits)
            if value < 27:
                input_to_output_counts.setdefault(value, {})[output_bits] = input_to_output_counts.setdefault(value, {}).get(output_bits, 0) + count

        input_to_output = {}
        for value, output_counter in input_to_output_counts.items():
            input_to_output[value] = max(output_counter, key=output_counter.get)

        if len(input_to_output) < 20:
            continue

        for little_trits in (False, True):
            for candidate_value in range(1, 27):
                candidate = normalize(int_to_vec(candidate_value, little_trits))
                if candidate == [0, 0, 0]:
                    continue

                score = 0
                seen_orbits = set()
                for value in range(27):
                    orbit = tuple(sorted(add_vec(value, candidate, scale, little_trits) for scale in range(3)))
                    if orbit in seen_orbits:
                        continue
                    seen_orbits.add(orbit)
                    labels = [input_to_output.get(item) for item in orbit]
                    if labels[0] is not None and labels[0] == labels[1] == labels[2]:
                        score += 6
                    else:
                        for left_index in range(3):
                            for right_index in range(left_index + 1, 3):
                                if labels[left_index] is not None and labels[left_index] == labels[right_index]:
                                    score += 1

                if score > best_score:
                    best_score = score
                    best_vector = candidate

    prediction = "".join(str(item) for item in best_vector)
    return prediction
'''
