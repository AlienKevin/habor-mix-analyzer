from qiskit import transpile

def run_and_analyze(circuit, aer_sim):
    transpiled = transpile(circuit, aer_sim)
    result = aer_sim.run(transpiled, shots=8192).result()
    counts = result.get_counts()

    def to_ternary(val, n=3):
        trits = []
        v = val
        for _ in range(n):
            trits.append(v % 3)
            v //= 3
        return trits[::-1]

    def from_ternary(trits):
        return "".join(str(t) for t in trits)

    def is_canonical(trits):
        for t in trits:
            if t == 1:
                return True
            if t == 2:
                return False
        return False

    def negate(trits):
        return tuple((-x) % 3 for x in trits)

    groups_a = {}
    groups_b = {}
    for bitstring, count in counts.items():
        bits = bitstring.replace(" ", "")
        if len(bits) != 10:
            continue
        out_str_a = bits[0:5]
        in_str_a = bits[5:10]
        x_val_a = int(in_str_a, 2)
        f_val_a = int(out_str_a, 2)
        if x_val_a < 27:
            if f_val_a not in groups_a:
                groups_a[f_val_a] = {}
            groups_a[f_val_a][x_val_a] = groups_a[f_val_a].get(x_val_a, 0) + count
        in_str_b = bits[5:10][::-1]
        out_str_b = bits[0:5][::-1]
        x_val_b = int(in_str_b, 2)
        f_val_b = int(out_str_b, 2)
        if x_val_b < 27:
            if f_val_b not in groups_b:
                groups_b[f_val_b] = {}
            groups_b[f_val_b][x_val_b] = groups_b[f_val_b].get(x_val_b, 0) + count

    def collect_votes(groups):
        votes = {}
        for f_val, xs in groups.items():
            x_list = list(xs.keys())
            for i in range(len(x_list)):
                for j in range(i + 1, len(x_list)):
                    x1 = x_list[i]
                    x2 = x_list[j]
                    t1 = to_ternary(x1)
                    t2 = to_ternary(x2)
                    d = tuple((t2[k] - t1[k]) % 3 for k in range(3))
                    if any(t != 0 for t in d):
                        w = xs[x1] + xs[x2]
                        votes[d] = votes.get(d, 0) + w
                        dn = negate(d)
                        votes[dn] = votes.get(dn, 0) + w
        return votes

    votes_a = collect_votes(groups_a)
    votes_b = collect_votes(groups_b)

    max_a = max(votes_a.values()) if votes_a else 0
    max_b = max(votes_b.values()) if votes_b else 0

    if max_a >= max_b:
        chosen_votes = votes_a
    else:
        chosen_votes = votes_b

    if chosen_votes:
        sorted_cands = sorted(chosen_votes.items(), key=lambda kv: -kv[1])
        top_vote = sorted_cands[0][1]
        threshold = top_vote * 0.5
        for cand, votes in sorted_cands:
            if votes < threshold:
                break
            if is_canonical(cand):
                return from_ternary(cand)
        return from_ternary(sorted_cands[0][0])

    return "001"
