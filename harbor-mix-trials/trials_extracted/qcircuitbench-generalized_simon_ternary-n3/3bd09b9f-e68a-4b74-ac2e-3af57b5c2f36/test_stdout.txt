from qiskit import transpile
import numpy as np

qasm_string = """OPENQASM 3.0;
include "stdgates.inc";

qubit[10] q;
bit[10] c;
h q[0];
h q[1];
h q[2];
h q[3];
h q[4];
Oracle q[0], q[1], q[2], q[3], q[4], q[5], q[6], q[7], q[8], q[9];
measure q[0] -> c[0];
measure q[1] -> c[1];
measure q[2] -> c[2];
measure q[3] -> c[3];
measure q[4] -> c[4];
measure q[5] -> c[5];
measure q[6] -> c[6];
measure q[7] -> c[7];
measure q[8] -> c[8];
measure q[9] -> c[9];
"""

def binary_to_ternary(value, length=3):
    """Convert integer to ternary string of given length."""
    digits = []
    for _ in range(length):
        digits.append(value % 3)
        value //= 3
    # digits are least significant first, reverse for most significant first
    return ''.join(str(d) for d in reversed(digits))

def ternary_to_int(s):
    """Convert ternary string to integer."""
    val = 0
    for ch in s:
        val = val * 3 + int(ch)
    return val

def run_and_analyze(circuit, aer_sim):
    # Transpile and run the circuit
    transpiled = transpile(circuit, aer_sim)
    result = aer_sim.run(transpiled, shots=1000).result()
    counts = result.get_counts()
    
    # Collect (x, f(x)) pairs
    pairs = []
    for bitstring, count in counts.items():
        # bitstring is in reverse order (qubit 0 is rightmost)
        # We measured all 10 qubits, so bitstring has 10 bits.
        # The rightmost 5 bits correspond to q0..q4 (input), the leftmost 5 bits correspond to q5..q9 (output).
        # Actually, bitstring is c9 c8 ... c0, so the rightmost 5 bits are c4..c0 (input), leftmost 5 bits are c9..c5 (output).
        # We'll extract input and output as integers.
        input_bits = bitstring[-5:]  # last 5 characters
        output_bits = bitstring[:5]  # first 5 characters
        x = int(input_bits, 2)
        fx = int(output_bits, 2)
        # Only consider x in range 0..26 (since 3^3=27)
        if x < 27:
            pairs.extend([(x, fx)] * count)
    
    # Group by f(x)
    groups = {}
    for x, fx in pairs:
        groups.setdefault(fx, []).append(x)
    
    # Find a group with exactly 3 distinct x values
    for fx, xs in groups.items():
        distinct_xs = list(set(xs))
        if len(distinct_xs) == 3:
            # Convert to ternary strings
            ternaries = [binary_to_ternary(x) for x in distinct_xs]
            # Sort to have a consistent order
            ternaries.sort()
            # Compute differences between first and second, first and third modulo 3
            a = [int(ch) for ch in ternaries[0]]
            b = [int(ch) for ch in ternaries[1]]
            c = [int(ch) for ch in ternaries[2]]
            # s = (b - a) mod 3
            s_digits = [(b[i] - a[i]) % 3 for i in range(3)]
            # Check if (c - a) mod 3 equals 2*s mod 3
            two_s_digits = [(2 * s_digits[i]) % 3 for i in range(3)]
            c_minus_a = [(c[i] - a[i]) % 3 for i in range(3)]
            if c_minus_a == two_s_digits:
                s_str = ''.join(str(d) for d in s_digits)
                if s_str != '000':
                    return s_str
    # If not found, return a default (maybe random)
    # We'll return '000' but that's invalid; we'll try to compute from any group with at least 2 distinct xs
    for fx, xs in groups.items():
        distinct_xs = list(set(xs))
        if len(distinct_xs) >= 2:
            ternaries = [binary_to_ternary(x) for x in distinct_xs[:2]]
            a = [int(ch) for ch in ternaries[0]]
            b = [int(ch) for ch in ternaries[1]]
            s_digits = [(b[i] - a[i]) % 3 for i in range(3)]
            if any(d != 0 for d in s_digits):
                return ''.join(str(d) for d in s_digits)
    # Fallback
    return '000'
