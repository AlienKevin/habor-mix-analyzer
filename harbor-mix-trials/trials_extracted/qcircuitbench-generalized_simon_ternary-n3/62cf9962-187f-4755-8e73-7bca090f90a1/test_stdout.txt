from qiskit import transpile
from collections import defaultdict
def run_and_analyze(circuit, aer_sim):
    tc = transpile(circuit, aer_sim)
    result = aer_sim.run(tc, shots=8192).result()
    counts = result.get_counts()
    def to_trits(x, n=3):
        t = []
        for _ in range(n):
            t.append(x % 3)
            x //= 3
        return t
    # Group inputs by output
    groups = defaultdict(lambda: defaultdict(int))
    for bs, cnt in counts.items():
        bs = bs.replace(' ', '')
        in_bits = bs[-5:]
        out_bits = bs[:-5]
        x_int = int(in_bits, 2)
        if x_int >= 27:
            continue
        groups[out_bits][x_int] += cnt
    # For each candidate s, count how many groups have x's that are consistent with shift s
    candidates = defaultdict(int)
    for out, xs in groups.items():
        x_list = list(xs.keys())
        if len(x_list) < 2:
            continue
        # try all pairs, compute trit-wise difference
        for i in range(len(x_list)):
            for j in range(len(x_list)):
                if i == j:
                    continue
                ti = to_trits(x_list[i])
                tj = to_trits(x_list[j])
                diff = tuple((b - a) % 3 for a, b in zip(ti, tj))
                if any(diff):
                    candidates[diff] += xs[x_list[i]] * xs[x_list[j]]
    # The most common nonzero difference should be s or 2s
    # Both s and 2s should appear with similar frequency. Pick the one with smallest 'value' representation
    if not candidates:
        return '000'
    # Group s and 2s together
    pairs = {}
    for d, w in candidates.items():
        d2 = tuple((2*x) % 3 for x in d)
        key = min(d, d2)
        pairs[key] = pairs.get(key, 0) + w
    best = max(pairs.items(), key=lambda kv: kv[1])[0]
    prediction = ''.join(str(t) for t in best)
    return prediction
"""
