from qiskit import transpile


def run_and_analyze(circuit, aer_sim):
    tcirc = transpile(circuit, aer_sim)
    result = aer_sim.run(tcirc, shots=8192).result()
    counts = result.get_counts()

    def clean_key(key):
        return key.replace(" ", "")

    def trits(value):
        out = []
        for _ in range(3):
            out.append(value % 3)
            value //= 3
        return out

    def normalize(vec):
        v = list(vec)
        for a in v:
            if a != 0:
                if a == 2:
                    v = [(2 * b) % 3 for b in v]
                break
        return v

    buckets = {}
    for key, cnt in counts.items():
        bits = clean_key(key)
        le = bits[::-1]
        if len(le) < 10:
            continue
        x_val = 0
        y_val = 0
        for i in range(5):
            if le[i] == "1":
                x_val += 1 << i
        for i in range(5):
            if le[5 + i] == "1":
                y_val += 1 << i
        if x_val < 27:
            if y_val not in buckets:
                buckets[y_val] = {}
            buckets[y_val][x_val] = buckets[y_val].get(x_val, 0) + cnt

    scores = {}
    for xs in buckets.values():
        vals = list(xs.keys())
        for i in range(len(vals)):
            for j in range(i + 1, len(vals)):
                a = vals[i]
                b = vals[j]
                ta = trits(a)
                tb = trits(b)
                diff = [((tb[k] - ta[k]) % 3) for k in range(3)]
                if diff == [0, 0, 0]:
                    continue
                diff = normalize(diff)
                tup = tuple(diff)
                scores[tup] = scores.get(tup, 0) + xs[a] * xs[b]

    if scores:
        best = max(scores, key=scores.get)
        prediction = "".join(str(x) for x in best)
    else:
        prediction = "100"
    return prediction
'''
