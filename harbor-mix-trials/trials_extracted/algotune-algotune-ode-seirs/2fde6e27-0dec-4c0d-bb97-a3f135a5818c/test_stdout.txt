--- Final Solver (used for performance test) ---
from __future__ import annotations

from typing import Any

import math
import numpy as np
from numba import njit


@njit(cache=True, fastmath=True)
def _seirs_rhs3(s: float, e: float, i: float, beta: float, sigma: float, gamma: float, omega: float):
    r = 1.0 - s - e - i
    inf = beta * s * i
    ds = -inf + omega * r
    de = inf - sigma * e
    di = sigma * e - gamma * i
    return ds, de, di


@njit(cache=True, fastmath=True)
def _rk4_fixed3(dt_total: float, s: float, e: float, i: float,
                beta: float, sigma: float, gamma: float, omega: float,
                hmax: float):
    if dt_total == 0.0:
        r = 1.0 - s - e - i
        return s, e, i, r
    direction = 1.0
    if dt_total < 0.0:
        direction = -1.0
        dt_total = -dt_total
    nsteps = int(math.ceil(dt_total / hmax))
    if nsteps < 1:
        nsteps = 1
    h = direction * (dt_total / nsteps)
    for _ in range(nsteps):
        k1s, k1e, k1i = _seirs_rhs3(s, e, i, beta, sigma, gamma, omega)
        k2s, k2e, k2i = _seirs_rhs3(s + 0.5*h*k1s, e + 0.5*h*k1e, i + 0.5*h*k1i, beta, sigma, gamma, omega)
        k3s, k3e, k3i = _seirs_rhs3(s + 0.5*h*k2s, e + 0.5*h*k2e, i + 0.5*h*k2i, beta, sigma, gamma, omega)
        k4s, k4e, k4i = _seirs_rhs3(s + h*k3s, e + h*k3e, i + h*k3i, beta, sigma, gamma, omega)
        s += (h/6.0) * (k1s + 2.0*k2s + 2.0*k3s + k4s)
        e += (h/6.0) * (k1e + 2.0*k2e + 2.0*k3e + k4e)
        i += (h/6.0) * (k1i + 2.0*k2i + 2.0*k3i + k4i)
    r = 1.0 - s - e - i
    return s, e, i, r


@njit(cache=True, fastmath=True)
def _dop853_like_dp5(dt_total: float, s: float, e: float, i: float,
                     beta: float, sigma: float, gamma: float, omega: float,
                     rtol: float, atol: float):
    # Dormand-Prince 5(4), integrated in the conserved (S,E,I) variables.
    if dt_total == 0.0:
        return s, e, i, 1.0 - s - e - i

    direction = 1.0
    if dt_total < 0.0:
        direction = -1.0
        dt_total = -dt_total

    max_rate = beta + sigma + gamma + omega
    if max_rate <= 1e-15:
        return s, e, i, 1.0 - s - e - i

    # Conservative initial step.  The controller rapidly increases it on easy cases.
    h_abs = min(dt_total, max(1e-4, 0.08 / max_rate))
    t = 0.0
    n = 0

    while t < dt_total and n < 200000:
        if t + h_abs > dt_total:
            h_abs = dt_total - t
        h = direction * h_abs

        k1s, k1e, k1i = _seirs_rhs3(s, e, i, beta, sigma, gamma, omega)

        y2s = s + h*(1.0/5.0)*k1s
        y2e = e + h*(1.0/5.0)*k1e
        y2i = i + h*(1.0/5.0)*k1i
        k2s, k2e, k2i = _seirs_rhs3(y2s, y2e, y2i, beta, sigma, gamma, omega)

        y3s = s + h*((3.0/40.0)*k1s + (9.0/40.0)*k2s)
        y3e = e + h*((3.0/40.0)*k1e + (9.0/40.0)*k2e)
        y3i = i + h*((3.0/40.0)*k1i + (9.0/40.0)*k2i)
        k3s, k3e, k3i = _seirs_rhs3(y3s, y3e, y3i, beta, sigma, gamma, omega)

        y4s = s + h*((44.0/45.0)*k1s + (-56.0/15.0)*k2s + (32.0/9.0)*k3s)
        y4e = e + h*((44.0/45.0)*k1e + (-56.0/15.0)*k2e + (32.0/9.0)*k3e)
        y4i = i + h*((44.0/45.0)*k1i + (-56.0/15.0)*k2i + (32.0/9.0)*k3i)
        k4s, k4e, k4i = _seirs_rhs3(y4s, y4e, y4i, beta, sigma, gamma, omega)

        y5s = s + h*((19372.0/6561.0)*k1s + (-25360.0/2187.0)*k2s + (64448.0/6561.0)*k3s + (-212.0/729.0)*k4s)
        y5e = e + h*((19372.0/6561.0)*k1e + (-25360.0/2187.0)*k2e + (64448.0/6561.0)*k3e + (-212.0/729.0)*k4e)
        y5i = i + h*((19372.0/6561.0)*k1i + (-25360.0/2187.0)*k2i + (64448.0/6561.0)*k3i + (-212.0/729.0)*k4i)
        k5s, k5e, k5i = _seirs_rhs3(y5s, y5e, y5i, beta, sigma, gamma, omega)

        y6s = s + h*((9017.0/3168.0)*k1s + (-355.0/33.0)*k2s + (46732.0/5247.0)*k3s + (49.0/176.0)*k4s + (-5103.0/18656.0)*k5s)
        y6e = e + h*((9017.0/3168.0)*k1e + (-355.0/33.0)*k2e + (46732.0/5247.0)*k3e + (49.0/176.0)*k4e + (-5103.0/18656.0)*k5e)
        y6i = i + h*((9017.0/3168.0)*k1i + (-355.0/33.0)*k2i + (46732.0/5247.0)*k3i + (49.0/176.0)*k4i + (-5103.0/18656.0)*k5i)
        k6s, k6e, k6i = _seirs_rhs3(y6s, y6e, y6i, beta, sigma, gamma, omega)

        y7s = s + h*((35.0/384.0)*k1s + (500.0/1113.0)*k3s + (125.0/192.0)*k4s + (-2187.0/6784.0)*k5s + (11.0/84.0)*k6s)
        y7e = e + h*((35.0/384.0)*k1e + (500.0/1113.0)*k3e + (125.0/192.0)*k4e + (-2187.0/6784.0)*k5e + (11.0/84.0)*k6e)
        y7i = i + h*((35.0/384.0)*k1i + (500.0/1113.0)*k3i + (125.0/192.0)*k4i + (-2187.0/6784.0)*k5i + (11.0/84.0)*k6i)
        k7s, k7e, k7i = _seirs_rhs3(y7s, y7e, y7i, beta, sigma, gamma, omega)

        # 4th-order embedded solution.
        z_s = s + h*((5179.0/57600.0)*k1s + (7571.0/16695.0)*k3s + (393.0/640.0)*k4s + (-92097.0/339200.0)*k5s + (187.0/2100.0)*k6s + (1.0/40.0)*k7s)
        z_e = e + h*((5179.0/57600.0)*k1e + (7571.0/16695.0)*k3e + (393.0/640.0)*k4e + (-92097.0/339200.0)*k5e + (187.0/2100.0)*k6e + (1.0/40.0)*k7e)
        z_i = i + h*((5179.0/57600.0)*k1i + (7571.0/16695.0)*k3i + (393.0/640.0)*k4i + (-92097.0/339200.0)*k5i + (187.0/2100.0)*k6i + (1.0/40.0)*k7i)

        # Weighted RMS error over S,E,I,R.  R error is minus sum of others.
        es = y7s - z_s
        ee = y7e - z_e
        ei = y7i - z_i
        er = -(es + ee + ei)
        old_r = 1.0 - s - e - i
        new_r = 1.0 - y7s - y7e - y7i
        scale_s = atol + rtol * max(abs(s), abs(y7s))
        scale_e = atol + rtol * max(abs(e), abs(y7e))
        scale_i = atol + rtol * max(abs(i), abs(y7i))
        scale_r = atol + rtol * max(abs(old_r), abs(new_r))
        err = math.sqrt((es/scale_s)*(es/scale_s) + (ee/scale_e)*(ee/scale_e) + (ei/scale_i)*(ei/scale_i) + (er/scale_r)*(er/scale_r)) * 0.5

        if err <= 1.0 or h_abs <= 1e-12:
            s = y7s
            e = y7e
            i = y7i
            t += h_abs
            if err == 0.0:
                fac = 5.0
            else:
                fac = 0.9 * err ** (-0.2)
                if fac < 0.2:
                    fac = 0.2
                elif fac > 5.0:
                    fac = 5.0
            h_abs *= fac
        else:
            fac = 0.9 * err ** (-0.25)
            if fac < 0.1:
                fac = 0.1
            elif fac > 0.5:
                fac = 0.5
            h_abs *= fac
        n += 1

    r = 1.0 - s - e - i
    return s, e, i, r



@njit(cache=True, fastmath=True)
def _equilibrium_state(beta: float, sigma: float, gamma: float, omega: float):
    # Disease-free equilibrium is (1,0,0,0) when R0 <= 1 or omega == 0.
    if beta <= gamma or omega <= 0.0 or sigma <= 0.0 or gamma <= 0.0:
        return 1.0, 0.0, 0.0, 0.0
    s = gamma / beta
    denom = gamma * (1.0 / sigma + 1.0 / beta) + gamma / omega
    i = (1.0 - s) / denom
    e = gamma * i / sigma
    r = gamma * i / omega
    return s, e, i, r


@njit(cache=True, fastmath=True)
def _is_close_to_equilibrium(s: float, e: float, i: float, beta: float, sigma: float, gamma: float, omega: float,
                             thresh: float):
    ds, de, di = _seirs_rhs3(s, e, i, beta, sigma, gamma, omega)
    r = 1.0 - s - e - i
    dr = gamma * i - omega * r
    scale = beta + sigma + gamma + omega + 1e-30
    return max(abs(ds), abs(de), abs(di), abs(dr)) <= thresh * scale

@njit(cache=True, fastmath=True)
def _solve_core(t0: float, t1: float, s: float, e: float, i: float, r: float,
                beta: float, sigma: float, gamma: float, omega: float):
    total = s + e + i + r
    if total != 0.0 and abs(total - 1.0) > 1e-14:
        s /= total
        e /= total
        i /= total
        r /= total

    dt = t1 - t0
    if dt == 0.0:
        return s, e, i, r

    max_rate = beta + sigma + gamma + omega
    if max_rate * abs(dt) < 1e-7:
        # One tiny RK4 step avoids controller overhead for essentially static cases.
        return _rk4_fixed3(dt, s, e, i, beta, sigma, gamma, omega, abs(dt))

    out_s, out_e, out_i, out_r = _dop853_like_dp5(dt, s, e, i, beta, sigma, gamma, omega, 2.0e-12, 1.0e-14)
    # If the trajectory has effectively reached a stable equilibrium, return the
    # exact equilibrium rather than an accumulated phase/tail integration error.
    if abs(dt) * max_rate > 300.0 and _is_close_to_equilibrium(out_s, out_e, out_i, beta, sigma, gamma, omega, 2e-8):
        eq_s, eq_e, eq_i, eq_r = _equilibrium_state(beta, sigma, gamma, omega)
        # Avoid snapping to a disease-free equilibrium while a visible epidemic tail remains.
        if not (eq_e == 0.0 and (out_e > 2e-7 or out_i > 2e-7)):
            return eq_s, eq_e, eq_i, eq_r
    return out_s, out_e, out_i, out_r


class Solver:
    def __init__(self) -> None:
        # Trigger numba compilation here; initialization time is not counted.
        _solve_core(0.0, 1.0, 0.89, 0.01, 0.005, 0.095, 0.35, 0.2, 0.1, 0.002)

    def solve(self, problem, **kwargs) -> Any:
        params = problem["params"]
        y0 = problem["y0"]
        s, e, i, r = _solve_core(
            float(problem["t0"]),
            float(problem["t1"]),
            float(y0[0]),
            float(y0[1]),
            float(y0[2]),
            float(y0[3]),
            float(params["beta"]),
            float(params["sigma"]),
            float(params["gamma"]),
            float(params["omega"]),
        )
        # Return the conserved final state.  Avoid clipping: allclose checks are stricter
        # than positivity, and the reference solver may contain tiny negative roundoff.
        return [float(s), float(e), float(i), float(r)]

# SciPy-compatible DOP853 implementation. Defined after Solver is fine because
# global lookup happens when solve() is called; __init__ compiles the function.
@njit(cache=True, fastmath=True)
def _norm4(a: float, b: float, c: float, d: float):
    return math.sqrt((a*a + b*b + c*c + d*d) * 0.25)


@njit(cache=True, fastmath=True)
def _rhs4_vals(s: float, e: float, i: float, r: float,
               beta: float, sigma: float, gamma: float, omega: float):
    inf = beta * s * i
    return -inf + omega * r, inf - sigma * e, sigma * e - gamma * i, gamma * i - omega * r


@njit(cache=True, fastmath=True)
def _dop853_scipy(dt_total: float, s: float, e: float, i: float, r: float,

    if dt_total == 0.0:
        return s, e, i, r
    direction = 1.0
    if dt_total < 0.0:
        direction = -1.0
        dt_total = -dt_total

    # Initial derivative and SciPy's select_initial_step(order=7).
    f0s, f0e, f0i, f0r = _rhs4_vals(s, e, i, r, beta, sigma, gamma, omega)
    scale_s = atol + abs(s) * rtol
    scale_e = atol + abs(e) * rtol
    scale_i = atol + abs(i) * rtol
    scale_r = atol + abs(r) * rtol
    d0 = _norm4(s/scale_s, e/scale_e, i/scale_i, r/scale_r)
    d1 = _norm4(f0s/scale_s, f0e/scale_e, f0i/scale_i, f0r/scale_r)
    if d0 < 1e-5 or d1 < 1e-5:
        h0 = 1e-6
    else:
        h0 = 0.01 * d0 / d1
    if h0 > dt_total:
        h0 = dt_total
    y1s = s + h0 * direction * f0s
    y1e = e + h0 * direction * f0e
    y1i = i + h0 * direction * f0i
    y1r = r + h0 * direction * f0r
    f1s, f1e, f1i, f1r = _rhs4_vals(y1s, y1e, y1i, y1r, beta, sigma, gamma, omega)
    d2 = _norm4((f1s-f0s)/scale_s, (f1e-f0e)/scale_e, (f1i-f0i)/scale_i, (f1r-f0r)/scale_r) / h0
    if d1 <= 1e-15 and d2 <= 1e-15:
        h1 = max(1e-6, h0 * 1e-3)

        h1 = (0.01 / max(d1, d2)) ** 0.125

    if h_abs <= 0.0:
        h_abs = dt_total

    n = 0
