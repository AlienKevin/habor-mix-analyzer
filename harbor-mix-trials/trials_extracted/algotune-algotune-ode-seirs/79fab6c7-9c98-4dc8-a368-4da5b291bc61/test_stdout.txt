import numpy as np
from typing import Any
import numba

# Numba JIT compiled RHS
@numba.njit(cache=True)
def _seirs_rhs(y, beta, sigma, gamma, omega):
    """SEIRS model RHS."""
    S, E, I, R = y[0], y[1], y[2], y[3]
    dS = -beta * S * I + omega * R
    dE = beta * S * I - sigma * E
    dI = sigma * E - gamma * I
    dR = gamma * I - omega * R
    return np.array([dS, dE, dI, dR])

# Numba JIT compiled RK45 step - Dormand-Prince method
@numba.njit(cache=True)
def _rk45_step(y, dt, beta, sigma, gamma, omega):
    """Perform one Dormand-Prince RK45 step. Returns (y_new, error_estimate)."""
    k1 = dt * _seirs_rhs(y, beta, sigma, gamma, omega)
    k2 = dt * _seirs_rhs(y + 0.25 * k1, beta, sigma, gamma, omega)
    k3 = dt * _seirs_rhs(y + 3.0/32.0 * k1 + 9.0/32.0 * k2, beta, sigma, gamma, omega)
    k4 = dt * _seirs_rhs(y + 1932.0/2197.0 * k1 - 7200.0/2197.0 * k2 + 7296.0/2197.0 * k3, beta, sigma, gamma, omega)
    k5 = dt * _seirs_rhs(y + 439.0/216.0 * k1 - 8.0 * k2 + 3680.0/513.0 * k3 - 845.0/4104.0 * k4, beta, sigma, gamma, omega)
    k6 = dt * _seirs_rhs(y - 8.0/27.0 * k1 + 2.0 * k2 - 3544.0/2565.0 * k3 + 1859.0/4104.0 * k4 - 11.0/40.0 * k5, beta, sigma, gamma, omega)
    
    # 5th order solution
    y_new = y + (16.0/135.0 * k1 + 6656.0/12825.0 * k3 + 28561.0/56430.0 * k4 - 9.0/50.0 * k5 + 2.0/55.0 * k6)
    
    # 4th order solution (for error estimation)
    y_4th = y + (25.0/216.0 * k1 + 1408.0/2565.0 * k3 + 2197.0/4104.0 * k4 - 1.0/5.0 * k5)
    
    error = np.max(np.abs(y_new - y_4th))
    return y_new, error

# Numba JIT compiled adaptive integrator
@numba.njit(cache=True)
def _integrate(y0, t0, t1, beta, sigma, gamma, omega):
    """Adaptive RK45 integration from t0 to t1."""
    rtol = 1e-8
    atol = 1e-10
    
    y = y0.copy()
    t = t0
    dt = 0.01  # Initial step size
    
    while t < t1:
        if t + dt > t1:
            dt = t1 - t
        
        y_new, error = _rk45_step(y, dt, beta, sigma, gamma, omega)
        
        # Error tolerance
        max_y = max(abs(y_new[0]), abs(y_new[1]), abs(y_new[2]), abs(y_new[3]))
        tol = atol + rtol * max_y
        
        if error <= tol:
            y = y_new
            t += dt
            # Increase step size for next step
            if error > 1e-16:
                dt *= min(5.0, 0.9 * (tol / error) ** 0.2)
        else:
            # Decrease step size and retry
            dt *= max(0.1, 0.9 * (tol / error) ** 0.25)
    
    return y

# Enforce conservation law and non-negativity
@numba.njit(cache=True)
def _enforce_conservation(y):
    """Ensure S + E + I + R = 1.0 and all values are non-negative."""
    # Clip negative values to zero
    if y[0] < 0: y[0] = 0.0
    if y[1] < 0: y[1] = 0.0
    if y[2] < 0: y[2] = 0.0
    if y[3] < 0: y[3] = 0.0
    
    # Normalize to sum to 1
    total = y[0] + y[1] + y[2] + y[3]
    if total > 0:
        y[0] /= total
        y[1] /= total
        y[2] /= total
        y[3] /= total
    return y


class Solver:
    def __init__(self):
        # Warmup JIT compilation - this time doesn't count toward runtime
        _integrate(np.array([0.9, 0.05, 0.03, 0.02]), 0.0, 0.1, 0.3, 0.2, 0.1, 0.01)
    
    def solve(self, problem, **kwargs) -> Any:
        t0 = problem["t0"]
        t1 = problem["t1"]
        y0 = np.array(problem["y0"])
        params = problem["params"]
        beta = params["beta"]
        sigma = params["sigma"]
        gamma = params["gamma"]
        omega = params["omega"]
        
        # Handle t1 <= t0 case
        if t1 <= t0:
            result = y0.copy()
        else:
            result = _integrate(y0, t0, t1, beta, sigma, gamma, omega)
        
        # Enforce conservation law and non-negativity
        result = _enforce_conservation(result)
        return result.tolist()
