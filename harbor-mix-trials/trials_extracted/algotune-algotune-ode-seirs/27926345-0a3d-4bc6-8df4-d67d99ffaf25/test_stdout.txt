import numpy as np
from scipy.integrate import solve_ivp
from typing import Any
from numba import njit
import warnings
warnings.filterwarnings('ignore')

# Pre-compile the derivative function with numba for speed
@njit(cache=True)
def _deriv_fast(y, beta, sigma, gamma, omega):
    S, E, I, R = y
    dS = -beta * S * I + omega * R
    dE = beta * S * I - sigma * E
    dI = sigma * E - gamma * I
    dR = gamma * I - omega * R
    return np.array([dS, dE, dI, dR])

class Solver:
    def __init__(self):
        # Initialize with JIT compilation
        _deriv_fast(np.array([0.9, 0.05, 0.03, 0.02]), 0.35, 0.2, 0.1, 0.002)
        self.method = 'RK45'
        self.rtol = 1e-6
        self.atol = 1e-9
    
    def solve(self, problem: dict[str, Any], **kwargs) -> list[float]:
        t0 = problem['t0']
        t1 = problem['t1']
        y0 = np.array(problem['y0'])
        params = problem['params']
        
        beta = params['beta']
        sigma = params['sigma']
        gamma = params['gamma']
        omega = params['omega']
        
        # Use solve_ivp with optimized settings and numba-compiled function
        sol = solve_ivp(
            fun=lambda t, y: _deriv_fast(y, beta, sigma, gamma, omega),
            t_span=(t0, t1),
            y0=y0,
            method=self.method,
            rtol=self.rtol,
            atol=self.atol
        )
        
        if sol.success:
            return sol.y[:, -1].tolist()
        else:
            raise RuntimeError(f"Solver failed: {sol.message}")
