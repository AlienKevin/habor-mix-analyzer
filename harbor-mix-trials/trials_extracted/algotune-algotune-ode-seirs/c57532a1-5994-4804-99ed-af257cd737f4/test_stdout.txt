import numpy as np
from scipy.integrate import odeint
from numba import njit, cfunc
from scipy import LowLevelCallable

@njit(cache=True, fastmath=True)
def seirs_odeint(y, t, beta, sigma, gamma, omega):
    S, E, I, R = y[0], y[1], y[2], y[3]
    bSI = beta * S * I
    sE = sigma * E
    gI = gamma * I
    wR = omega * R
    return np.array([-bSI + wR, bSI - sE, sE - gI, gI - wR])

@njit(cache=True, fastmath=True)
def jac_odeint(y, t, beta, sigma, gamma, omega):
    S, E, I, R = y[0], y[1], y[2], y[3]
    J = np.zeros((4,4))
    J[0,0] = -beta*I
    J[0,2] = -beta*S
    J[0,3] = omega
    J[1,0] = beta*I
    J[1,1] = -sigma
    J[1,2] = beta*S
    J[2,1] = sigma
    J[2,2] = -gamma
    J[3,2] = gamma
    J[3,3] = -omega
    return J

_y = np.array([0.9, 0.05, 0.03, 0.02])
seirs_odeint(_y, 0.0, 0.35, 0.2, 0.1, 0.002)
jac_odeint(_y, 0.0, 0.35, 0.2, 0.1, 0.002)

class Solver:
    def solve(self, problem, **kwargs):
        t0 = problem["t0"]
        t1 = problem["t1"]
        y0 = np.array(problem["y0"], dtype=np.float64)
        p = problem["params"]
        args = (p["beta"], p["sigma"], p["gamma"], p["omega"])
        sol = odeint(seirs_odeint, y0, [t0, t1], args=args, Dfun=jac_odeint, rtol=1e-10, atol=1e-10, full_output=False, mxstep=100000)
        return sol[-1].tolist()
