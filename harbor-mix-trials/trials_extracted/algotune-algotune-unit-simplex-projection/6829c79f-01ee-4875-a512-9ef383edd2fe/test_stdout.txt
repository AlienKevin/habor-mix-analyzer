import numpy as np
from numba import njit
from typing import Any


@njit(cache=True)
def _project_simplex_numba(y: np.ndarray) -> np.ndarray:
    """Numba-compiled simplex projection - no intermediate arrays."""
    n = y.shape[0]
    sorted_y = np.sort(y)[::-1]

    cumsum = 0.0
    rho = 0
    for j in range(n):
        cumsum += sorted_y[j]
        if sorted_y[j] * (j + 1) > cumsum - 1.0:
            rho = j

    cumsum_rho = 0.0
    for j in range(rho + 1):
        cumsum_rho += sorted_y[j]
    theta = (cumsum_rho - 1.0) / (rho + 1)

    x = np.empty(n, dtype=np.float64)
    for i in range(n):
        val = y[i] - theta
        x[i] = val if val > 0.0 else 0.0
    return x


@njit(cache=True)
def _find_rho_and_project(sorted_y: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Find rho and project - avoids intermediate arrays for large inputs."""
    n = y.shape[0]

    cumsum = 0.0
    rho = 0
    for j in range(n):
        cumsum += sorted_y[j]
        if sorted_y[j] * (j + 1) > cumsum - 1.0:
            rho = j

    cumsum_rho = 0.0
    for j in range(rho + 1):
        cumsum_rho += sorted_y[j]
    theta = (cumsum_rho - 1.0) / (rho + 1)

    x = np.empty(n, dtype=np.float64)
    for i in range(n):
        val = y[i] - theta
        x[i] = val if val > 0.0 else 0.0
    return x


class Solver:
    def __init__(self):
        warmup = np.array([1.0, 1.2, 0.5])
        _project_simplex_numba(warmup)
        _find_rho_and_project(np.sort(warmup)[::-1], warmup)

    def solve(self, problem: dict[str, Any], **kwargs) -> dict[str, Any]:
        y = np.asarray(problem["y"], dtype=np.float64).ravel()
        n = len(y)

        if n <= 2000:
            # Fully numba-compiled for small/medium arrays
            x = _project_simplex_numba(y)
        else:
            # numpy sort (highly optimized C) + numba for post-sort
            sorted_y = np.sort(y)[::-1]
            x = _find_rho_and_project(sorted_y, y)

        return {"solution": x}
