import numpy as np
from numba import njit
from typing import Any

@njit("float64[:](float64[:])", cache=True)
def project_simplex(v):
    U = v.copy()
    s = 0.0
    rho = 0
    z = 1.0
    
    left = 0
    right = len(U) - 1
    
    while left <= right:
        # Pick a random pivot
        k = np.random.randint(left, right + 1)
        c = U[k]
        
        # 3-way partition
        i = left
        lt = left
        gt = right
        
        while i <= gt:
            if U[i] > c:
                U[i], U[gt] = U[gt], U[i]
                gt -= 1
            elif U[i] < c:
                U[i], U[lt] = U[lt], U[i]
                lt += 1
                i += 1
            else:
                i += 1
                
        ds = 0.0
        drho = 0
        ds_E = 0.0
        drho_E = 0
        
        # Elements > c are from gt + 1 to right
        for idx in range(gt + 1, right + 1):
            drho += 1
            ds += U[idx]
            
        # Elements == c are from lt to gt
        for idx in range(lt, gt + 1):
            drho_E += 1
            ds_E += U[idx]
                
        if (s + ds) - (rho + drho) * c < z:
            s += ds
            rho += drho
            if (s + ds_E) - (rho + drho_E) * c < z:
                s += ds_E
                rho += drho_E
                # true threshold < c, but since we committed == c,
                # we must search for smaller elements
                right = lt - 1
            else:
                # true threshold is exactly c, we are done
                break
        else:
            # true threshold >= c, search for larger elements
            left = gt + 1

    theta = (s - z) / rho
    x = np.empty_like(v)
    for i in range(len(v)):
        val = v[i] - theta
        x[i] = val if val > 0 else 0.0
    return x

class Solver:
    def __init__(self):
        # Warmup the JIT to ensure it doesn't compile during the first call
        project_simplex(np.array([1.0, 2.0], dtype=np.float64))

    def solve(self, problem: dict[str, Any], **kwargs) -> dict[str, Any]:
        y = np.asarray(problem["y"], dtype=np.float64).ravel()
        x = project_simplex(y)
        return {"solution": x}
