import numpy as np
from typing import Any


class Solver:
    """
    Solver for Euclidean projection onto the probability simplex.
    
    Solves: minimize_x (1/2) ||x - y||^2 subject to x^T 1 = 1, x >= 0
    
    Uses the O(n log n) algorithm from https://arxiv.org/pdf/1309.1541
    """
    
    def __init__(self):
        pass
    
    def solve(self, problem, **kwargs) -> dict:
        """
        Project point y onto the probability simplex.
        
        :param problem: Dictionary with key "y" containing the vector y (list or numpy array)
        :return: Dictionary with key "solution" containing the optimal x (numpy array)
        """
        # Convert input to numpy array
        y = np.asarray(problem.get("y"), dtype=np.float64)
        y = np.atleast_1d(y)
        n = y.shape[0]
        
        if n == 0:
            return {"solution": y.copy()}
        
        # Sort y in descending order
        sorted_y = np.flip(np.sort(y))
        
        # Compute cumulative sum minus 1
        cumsum_m1 = np.cumsum(sorted_y) - 1.0
        
        # Find rho: the largest index where sorted_y[i] > cumsum_m1[i] / (i+1)
        # Condition: sorted_y[i] * (i+1) > cumsum_m1[i]
        # Since sorted_y is sorted descending, valid[i] is True for i <= rho
        # and False for i > rho (monotonic property)
        
        # Vectorized check
        i = np.arange(1, n + 1, dtype=np.float64)
        valid = sorted_y * i > cumsum_m1
        
        if np.any(valid):
            # Find last valid index using count_nonzero (exploiting monotonicity)
            rho = np.count_nonzero(valid) - 1
            theta = cumsum_m1[rho] / (rho + 1.0)
        else:
            # All values are very negative - uniform solution x_i = 1/n
            theta = (np.sum(y) - 1.0) / n
        
        # Project onto the simplex: x = max(y - theta, 0)
        x = np.maximum(y - theta, 0.0)
        
        return {"solution": x}
