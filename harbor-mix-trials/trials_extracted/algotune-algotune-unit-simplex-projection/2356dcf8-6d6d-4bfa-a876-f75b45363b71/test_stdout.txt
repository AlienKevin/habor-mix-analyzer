import numpy as np

class Solver:
    def solve(self, problem, **kwargs):
        y = np.asarray(problem["y"], dtype=np.float64).ravel()
        n = y.size
        s = np.sort(y)[::-1]
        cs = np.cumsum(s) - 1.0
        idx = np.arange(1, n + 1)
        cond = s * idx > cs
        rho = np.nonzero(cond)[0][-1]
        theta = cs[rho] / (rho + 1)
        x = y - theta
        np.maximum(x, 0, out=x)
        return {"solution": x}
--- End Solver ---
