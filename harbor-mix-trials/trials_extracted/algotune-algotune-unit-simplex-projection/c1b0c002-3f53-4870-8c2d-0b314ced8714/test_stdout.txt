import numpy as np
from numba import njit


@njit(cache=True)
def _project_simplex(y):
    n = len(y)

    s = 0.0
    for i in range(n):
        s += y[i]
    theta = (s - 1.0) / n

    for _ in range(n):
        s = 0.0
        k = 0
        for i in range(n):
            if y[i] > theta:
                s += y[i]
                k += 1

        if k == 0:
            break

        new_theta = (s - 1.0) / k

        if new_theta == theta:
            break

        theta = new_theta

    x = np.empty(n)
    for i in range(n):
        v = y[i] - theta
        x[i] = v if v > 0.0 else 0.0

    return x


class Solver:
    def __init__(self):
        _project_simplex(np.array([1.0, 2.0]))

    def solve(self, problem, **kwargs):
        y = np.asarray(problem.get("y"), dtype=np.float64).ravel()
        x = _project_simplex(y)
        return {"solution": x}--- End Solver ---
