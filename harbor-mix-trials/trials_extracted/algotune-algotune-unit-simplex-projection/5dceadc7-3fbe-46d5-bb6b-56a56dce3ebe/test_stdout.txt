import numpy as np
from typing import Any


class Solver:
    def solve(self, problem: dict[str, Any], **kwargs) -> dict[str, Any]:
        """
        Solve the Euclidean projection onto the probability simplex.

        Uses an O(n log n) algorithm based on sorted projection.
        Reference: https://arxiv.org/pdf/1309.1541

        :param problem: Dictionary with key "y" containing the input vector
        :return: Dictionary with key "solution" containing the projected vector
        """
        y = problem.get("y")

        if y is None:
            return {"solution": np.array([], dtype=np.float64)}

        y = np.asarray(y, dtype=np.float64).flatten()
        n = len(y)

        # Handle edge cases
        if n == 0:
            return {"solution": np.array([], dtype=np.float64)}

        if n == 1:
            return {"solution": np.array([1.0], dtype=np.float64)}

        # Sort y in descending order
        sorted_y = np.sort(y)[::-1]

        # Compute cumulative sum: cumsum(sorted_y) - 1
        cumsum_sorted = np.cumsum(sorted_y)
        cumsum_sorted -= 1.0

        # Find rho: largest index where sorted_y[i] > cumsum_sorted[i] / (i + 1)
        # Vectorized computation of thresholds
        thresholds = cumsum_sorted / np.arange(1, n + 1, dtype=np.float64)

        # Find the last index where sorted_y > threshold
        valid_mask = sorted_y > thresholds
        if not valid_mask.any():
            rho = 0
            theta = 0.0
        else:
            rho = np.where(valid_mask)[0][-1]
            theta = cumsum_sorted[rho] / (rho + 1)

        # Project: x = max(y - theta, 0)
        solution = np.maximum(y - theta, 0.0)

        return {"solution": solution}
