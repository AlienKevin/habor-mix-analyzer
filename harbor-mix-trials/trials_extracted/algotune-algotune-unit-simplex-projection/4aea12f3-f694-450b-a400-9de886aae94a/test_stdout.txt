import numpy as np
from numba import njit
from typing import Any

@njit(cache=True)
def _project_simplex_numba(y: np.ndarray) -> np.ndarray:
    """JIT-compiled Euclidean projection onto probability simplex."""
    n = len(y)
    sorted_y = np.sort(y)[::-1]
    cumsum_y = np.cumsum(sorted_y) - 1.0
    
    rho = 0
    for i in range(n):
        if sorted_y[i] > cumsum_y[i] / (i + 1):
            rho = i
    
    theta = cumsum_y[rho] / (rho + 1)
    x = np.maximum(y - theta, 0.0)
    return x

class Solver:
    def solve(self, problem: dict[str, Any], **kwargs) -> dict[str, Any]:
        """
        Solve Euclidean projection onto probability simplex.
        
        Args:
            problem: dict with key "y" containing the vector to project
            **kwargs: additional arguments (unused)
        
        Returns:
            dict with key "solution" containing the projected vector
        """
        y = np.asarray(problem.get("y"), dtype=np.float64)
        solution = _project_simplex_numba(y)
        return {"solution": solution}
