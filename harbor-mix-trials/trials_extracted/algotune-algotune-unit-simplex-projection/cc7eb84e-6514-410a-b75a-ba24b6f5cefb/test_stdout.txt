import numpy as np
from numba import njit


@njit(cache=True)
def _proj_loop(y, s_asc):
    """Post-sort: find rho, compute theta, project. s_asc = y sorted ascending."""
    n = y.shape[0]
    cs = 0.0
    rho = 0
    cs_rho = 0.0
    idx = n - 1
    for i in range(n):
        val = s_asc[idx]
        cs += val
        if val > (cs - 1.0) / (i + 1):
            rho = i
            cs_rho = cs
        idx -= 1
    theta = (cs_rho - 1.0) / (rho + 1)
    return np.maximum(y - theta, 0.0)


@njit(cache=True)
def _solve_bisect(y):
    """Binary search for theta (O(n) per iter) + correction step."""
    n = y.shape[0]

    lo = y[0]
    hi = y[0]
    for i in range(1, n):
        if y[i] < lo:
            lo = y[i]
        if y[i] > hi:
            hi = y[i]
    lo -= 1.0

    for _ in range(35):
        mid = (lo + hi) * 0.5
        s = 0.0
        for i in range(n):
            diff = y[i] - mid
            if diff > 0.0:
                s += diff
        if s > 1.0:
            lo = mid
        else:
            hi = mid

    theta = hi
    result = np.empty(n)
    total = 0.0
    pos_count = 0
    for i in range(n):
        diff = y[i] - theta
        if diff > 0.0:
            result[i] = diff
            total += diff
            pos_count += 1
        else:
            result[i] = 0.0

    if pos_count > 0:
        correction = (total - 1.0) / pos_count
        if correction != 0.0:
            theta += correction
            for i in range(n):
                diff = y[i] - theta
                result[i] = diff if diff > 0.0 else 0.0

    return result


class Solver:
    def __init__(self):
        _proj_loop(np.array([1.0, 2.0, 3.0]), np.sort(np.array([1.0, 2.0, 3.0])))
        _solve_bisect(np.array([1.0, 2.0, 3.0]))

    def solve(self, problem, **kwargs):
        y = np.array(problem["y"], dtype=np.float64)
        n = y.shape[0]
        if n < 30000:
            s = np.sort(y)
            x = _proj_loop(y, s)
        else:
            x = _solve_bisect(y)
        return {"solution": x}
