gate oracle q0, q1, q2, q3, q4, q5 {
    // Set q[4] and q[5] to |10⟩ (representing base-3 value 2)
    x q[5];
    // Determine x₁ (second trit) from q[1], q[2]
    // x₁ = 0: |00⟩, x₁ = 1: |01⟩, x₁ = 2: |10⟩
    // Helper: set q[3] = 1 when x₁ = 2
    x q[1];
    x q[2];
    ccx q[1], q[2], q[3];
    x q[1];
    x q[2];
    // Compute f(x)₁ = (x₀ + 2) mod 3
    // CNOT: q[0] controls q[4]

    ccx q[0], q[3], q[5];
