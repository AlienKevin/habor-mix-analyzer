import numpy as np
from scipy import signal, linalg
import numba
from typing import Any

@numba.njit(cache=True)
def simulate_loop(xout, U, Ad, Bd0, Bd1, n_steps):
    for i in range(1, n_steps):
        xout[i] = xout[i-1] @ Ad + U[i-1] @ Bd0 + U[i] @ Bd1
    return xout

class Solver:
    def __init__(self):
        # Pre-compile the numba function with a dummy call
        dummy_xout = np.zeros((2, 2), dtype=np.float64)
        dummy_U = np.zeros((2, 1), dtype=np.float64)
        dummy_Ad = np.eye(2, dtype=np.float64)
        dummy_Bd0 = np.zeros((2, 2), dtype=np.float64)
        dummy_Bd1 = np.zeros((2, 2), dtype=np.float64)
        # Shapes must match what simulate_loop expects
        dummy_xout = np.zeros((2, 2), dtype=np.float64)
        dummy_U = np.zeros((2, 1), dtype=np.float64)
        dummy_Ad = np.zeros((2, 2), dtype=np.float64)
        dummy_Bd0 = np.zeros((1, 2), dtype=np.float64)
        dummy_Bd1 = np.zeros((1, 2), dtype=np.float64)
        simulate_loop(dummy_xout, dummy_U, dummy_Ad, dummy_Bd0, dummy_Bd1, 2)

    def solve(self, problem: dict[str, Any], **kwargs) -> dict[str, list[float]]:
        num = problem["num"]
        den = problem["den"]
        u = np.array(problem["u"], dtype=np.float64)
        t = np.array(problem["t"], dtype=np.float64)

        n_steps = len(t)
        if n_steps == 0:
            return {"yout": []}
        elif n_steps == 1:
            # Edge case, though usually t has >1 elements
            A, B, C, D = signal.tf2ss(num, den)
            yout = u[0] * D[0, 0]
            return {"yout": [float(yout)]}

        dt = t[1] - t[0]

        A, B, C, D = signal.tf2ss(num, den)
        n_states = A.shape[0]
        n_inputs = B.shape[1]

        # Exact discretization with First-Order Hold (linear interpolation)
        M = np.vstack([
            np.hstack([A * dt, B * dt, np.zeros((n_states, n_inputs))]),
            np.hstack([np.zeros((n_inputs, n_states + n_inputs)), np.identity(n_inputs)]),
            np.zeros((n_inputs, n_states + 2 * n_inputs))
        ])
        expMT = linalg.expm(M.T)
        
        Ad = np.ascontiguousarray(expMT[:n_states, :n_states])
        Bd1 = np.ascontiguousarray(expMT[n_states+n_inputs:, :n_states])
        Bd0 = np.ascontiguousarray(expMT[n_states:n_states + n_inputs, :n_states] - Bd1)

        xout = np.zeros((n_steps, n_states), dtype=np.float64)
        U = np.ascontiguousarray(u.reshape(-1, 1))

        xout = simulate_loop(xout, U, Ad, Bd0, Bd1, n_steps)
        
        yout = (xout @ C.T).squeeze() + (U @ D.T).squeeze()
        
        # Ensure yout is a 1D array/list
        if yout.ndim == 0:
            yout = np.array([yout])
            
        return {"yout": yout.tolist()}
