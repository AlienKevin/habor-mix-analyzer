import numpy as np
from scipy.signal import tf2ss, lsim, lti
from scipy.linalg import expm
from numba import njit


@njit(cache=True, boundscheck=False, fastmath=False)
def _simulate_uniform(Ad, Bd0, Bd1, C_vec, D_val, u, n_steps, n):
    yout = np.empty(n_steps)
    x = np.zeros(n)
    new_x = np.empty(n)
    yout[0] = D_val * u[0]
    for i in range(1, n_steps):
        u_prev = u[i - 1]
        u_curr = u[i]
        for k in range(n):
            s = Bd0[k] * u_prev + Bd1[k] * u_curr
            for j in range(n):
                s += Ad[k, j] * x[j]
            new_x[k] = s
        for k in range(n):
            x[k] = new_x[k]
        s = D_val * u_curr
        for k in range(n):
            s += C_vec[k] * x[k]
        yout[i] = s
    return yout


def _to_float_array(x):
    """Convert input to a contiguous float64 array with minimal overhead."""
    if isinstance(x, np.ndarray):
        if x.dtype == np.float64 and x.flags['C_CONTIGUOUS']:
            return x
        return np.ascontiguousarray(x, dtype=np.float64)
    return np.fromiter(x, dtype=np.float64, count=len(x))


class Solver:
    def __init__(self):
        # Trigger JIT compilation (does not count toward solve time)
        Ad = np.eye(2)
        Bd0 = np.zeros(2)
        Bd1 = np.zeros(2)
        C_vec = np.zeros(2)
        u = np.zeros(2)
        _simulate_uniform(Ad, Bd0, Bd1, C_vec, 0.0, u, 2, 2)

    def solve(self, problem, **kwargs):
        num = problem["num"]
        den = problem["den"]
        u = _to_float_array(problem["u"])
        t = _to_float_array(problem["t"])

        n_steps = u.shape[0]
        if n_steps == 0:
            return {"yout": []}

        A, B, C, D = tf2ss(num, den)
        n = A.shape[0]
        D_val = float(D.flat[0]) if D.size > 0 else 0.0

        # Pure-gain case (no states)
        if n == 0:
            return {"yout": (D_val * u).tolist()}

        if n_steps == 1:
            return {"yout": [D_val * float(u[0])]}

        # Cheap uniform-spacing check using endpoints (with one diff sanity probe)
        dt = float(t[1] - t[0])
        if n_steps > 2:
            expected_last = t[0] + dt * (n_steps - 1)
            uniform = abs(float(t[-1]) - expected_last) <= 1e-10 * max(abs(expected_last), 1.0)
            # Sanity-check a midpoint to catch non-monotonic spacing that still hits the endpoint.
            if uniform and n_steps > 4:
                mid = n_steps // 2
                expected_mid = t[0] + dt * mid
                uniform = abs(float(t[mid]) - expected_mid) <= 1e-10 * max(abs(expected_mid), 1.0)
        else:
            uniform = True

        if uniform:
            # Build augmented matrix for first-order hold (FOH) discretization.
            # Mirrors scipy.signal.lsim: dt is baked into A and B blocks only;
            # the identity coupling block stays as I.
            M = np.zeros((n + 2, n + 2))
            M[:n, :n] = A * dt
            M[:n, n] = B[:, 0] * dt
            M[n, n + 1] = 1.0
            expM = expm(M)

            Ad = np.ascontiguousarray(expM[:n, :n])
            col_n = expM[:n, n]
            col_n1 = expM[:n, n + 1]
            Bd1_col = np.ascontiguousarray(col_n1)
            Bd0_col = np.ascontiguousarray(col_n - col_n1)
            C_vec = np.ascontiguousarray(C[0])

            yout = _simulate_uniform(Ad, Bd0_col, Bd1_col, C_vec, D_val, u, n_steps, n)
            return {"yout": yout.tolist()}

        # Fallback: scipy's lsim handles non-uniform case
        system = lti(num, den)
        _, yout, _ = lsim(system, u, t)
        return {"yout": np.asarray(yout).tolist()}
