import numpy as np
from numba import jit
from scipy.linalg import expm
from scipy.signal import tf2ss


@jit(nopython=True, cache=True)
def _simulate_foh(Ad, Bd0_vec, Bd1_vec, C_vec, D_val, u, y):
    n_states = Ad.shape[0]
    n_steps = len(u)
    x = np.zeros(n_states)
    x_next = np.zeros(n_states)
    for i in range(1, n_steps):
        y[i - 1] = D_val * u[i - 1]
        for j in range(n_states):
            y[i - 1] += C_vec[j] * x[j]
        for j in range(n_states):
            acc = Bd0_vec[j] * u[i - 1] + Bd1_vec[j] * u[i]
            for k in range(n_states):
                acc += Ad[j, k] * x[k]
            x_next[j] = acc
        tmp = x
        x = x_next
        x_next = tmp
    y[-1] = D_val * u[-1]
    for j in range(n_states):
        y[-1] += C_vec[j] * x[j]


class Solver:
    def __init__(self):
        # Warm up numba JIT
        n_states = 2
        Ad = np.eye(n_states)
        Bd0 = np.ones(n_states)
        Bd1 = np.ones(n_states)
        C_vec = np.ones(n_states)
        D_val = 0.0
        u = np.ones(10)
        y = np.empty(10)
        _simulate_foh(Ad, Bd0, Bd1, C_vec, D_val, u, y)

    def solve(self, problem: dict, **kwargs) -> dict:
        num = np.asarray(problem["num"], dtype=np.float64)
        den = np.asarray(problem["den"], dtype=np.float64)
        u = np.asarray(problem["u"], dtype=np.float64)
        t = np.asarray(problem["t"], dtype=np.float64)

        n = len(t)
        if n == 0:
            return {"yout": []}
        if n == 1:
            A, B, C, D = tf2ss(num, den)
            D_val = float(D.ravel()[0])
            return {"yout": [float(D_val * u[0])]}

        dt = t[1] - t[0]
        if not np.allclose(np.diff(t), dt, rtol=1e-10, atol=1e-15):
            raise ValueError("Time steps must be equally spaced")

        A, B, C, D = tf2ss(num, den)

        n_states = A.shape[0]
        m = B.shape[1]

        M = np.vstack(
            [
                np.hstack([A * dt, B * dt, np.zeros((n_states, m))]),
                np.hstack([np.zeros((m, n_states + m)), np.eye(m)]),
                np.zeros((m, n_states + 2 * m)),
            ]
        )
        expMT = expm(M.T)
        Ad = np.ascontiguousarray(expMT[:n_states, :n_states].T)
        Bd1 = np.ascontiguousarray(expMT[n_states + m :, :n_states].T)
        Bd0 = np.ascontiguousarray(expMT[n_states : n_states + m, :n_states].T - Bd1)

        C_vec = np.ascontiguousarray(C.ravel())
        D_val = float(D.ravel()[0])
        Bd0_v = np.ascontiguousarray(Bd0.ravel())
        Bd1_v = np.ascontiguousarray(Bd1.ravel())

        y = np.empty(n, dtype=np.float64)
        _simulate_foh(Ad, Bd0_v, Bd1_v, C_vec, D_val, u, y)

        return {"yout": y.tolist()}
