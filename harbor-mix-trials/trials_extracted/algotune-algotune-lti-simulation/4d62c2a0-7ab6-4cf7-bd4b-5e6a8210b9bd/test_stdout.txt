import numpy as np
from scipy import signal
from scipy.signal import lti
from scipy.linalg import expm
from numba import njit
from typing import Any


@njit(cache=True)
def simulate_lti(U, Ad, Bd0, Bd1, C, D):
    """
    Numba-accelerated LTI simulation using linear interpolation.

    Uses the same algorithm as scipy.signal.lsim with interp=True.
    State update: x[i] = x[i-1] @ Ad.T + u[i-1] @ Bd0.T + u[i] @ Bd1.T
    Output: y[i] = x[i] @ C.T + u[i] @ D.T
    """
    n_steps = U.shape[0]
    n_states = Ad.shape[0]
    n_inputs = U.shape[1]

    xout = np.zeros((n_steps, n_states))

    for i in range(1, n_steps):
        for j in range(n_states):
            val = 0.0
            for k in range(n_states):
                val += xout[i-1, k] * Ad[k, j]
            for k in range(n_inputs):
                val += U[i-1, k] * Bd0[k, j]
                val += U[i, k] * Bd1[k, j]
            xout[i, j] = val

    yout = np.zeros(n_steps)
    for i in range(n_steps):
        val = 0.0
        for j in range(n_states):
            val += xout[i, j] * C[0, j]
        for k in range(n_inputs):
            val += U[i, k] * D[0, k]
        yout[i] = val

    return yout


class Solver:
    def solve(self, problem: dict, **kwargs) -> dict:
        """
        Solves the LTI simulation problem using numba-accelerated computation.

        Uses the same algorithm as scipy.signal.lsim with linear interpolation
        but with precomputed state transition matrices and numba JIT compilation
        for significant speedup.

        :param problem: A dictionary with keys:
            - "num": numerator polynomial coefficients (highest power first)
            - "den": denominator polynomial coefficients (highest power first)
            - "u": input signal values
            - "t": time points (monotonically increasing, equally spaced)
        :return: A dictionary with key "yout" containing the output signal values
        """
        num = problem["num"]
        den = problem["den"]
        u = np.asarray(problem["u"], dtype=np.float64)
        t = np.asarray(problem["t"], dtype=np.float64)

        # Handle single time point case
        if len(t) == 1:
            yout = np.array([0.0])  # Initial output is 0 for single point
            return {"yout": yout.tolist()}

        # Convert to state-space representation
        sys = lti(num, den)._as_ss()
        A, B, C, D = map(np.asarray, (sys.A, sys.B, sys.C, sys.D))

        n_states = A.shape[0]
        n_inputs = B.shape[1]

        # Time step
        dt = t[1] - t[0]

        # Build and exponentiate the extended matrix for linear interpolation
        # This is the same algorithm used in scipy.signal.lsim
        M = np.vstack([
            np.hstack([A * dt, B * dt, np.zeros((n_states, n_inputs))]),
            np.hstack([np.zeros((n_inputs, n_states + n_inputs)), np.identity(n_inputs)]),
            np.zeros((n_inputs, n_states + 2 * n_inputs))
        ])

        expMT = expm(M.T)
        Ad = expMT[:n_states, :n_states]
        Bd1 = expMT[n_states + n_inputs:, :n_states]
        Bd0 = expMT[n_states:n_states + n_inputs, :n_states] - Bd1

        # Prepare input array for numba (needs 2D shape: n_steps x n_inputs)
        if u.ndim == 1:
            u = u[:, np.newaxis]

        # Run simulation
        yout = simulate_lti(u, Ad, Bd0, Bd1, C, D)

        return {"yout": yout.tolist()}--- End Solver ---
