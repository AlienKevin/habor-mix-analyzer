@numba.njit(cache=True, fastmath=True)
def _sim1(Ad, Bd0, Bd1, Cf, u, Dv):
    ns = len(u)
    a0 = Ad[0,0]; b00 = Bd0[0]; b10 = Bd1[0]; c0 = Cf[0]
    x0 = 0.0
    y = np.empty(ns)
    y[0] = Dv * u[0]
    for i in range(1, ns):
        x0 = a0*x0 + b00*u[i-1] + b10*u[i]
        y[i] = c0*x0 + Dv*u[i]
    return y

@numba.njit(cache=True, fastmath=True)
def _sim_gen(Ad, Bd0, Bd1, Cf, u, Dv):
    ns = len(u)
    nx = Ad.shape[0]
    x = np.zeros(nx)
    y = np.empty(ns)
    y[0] = Dv * u[0]
    for i in range(1, ns):
        x = Ad @ x + Bd0 * u[i-1] + Bd1 * u[i]
        y[i] = Cf @ x + Dv * u[i]
    return y

class Solver:
    def solve(self, problem, **kwargs):
        num = np.asarray(problem["num"], dtype=np.float64)
        den = np.asarray(problem["den"], dtype=np.float64)
        u = np.asarray(problem["u"], dtype=np.float64)
        t = np.asarray(problem["t"], dtype=np.float64)
        ns = len(t)
        A, B, C, D = signal.tf2ss(num, den)
        A = np.ascontiguousarray(A, dtype=np.float64)
        B = np.ascontiguousarray(B, dtype=np.float64)
        C = np.ascontiguousarray(C, dtype=np.float64)
        D = np.ascontiguousarray(D, dtype=np.float64)
        nx = A.shape[0]
        Dv = float(D.ravel()[0]) if D.size > 0 else 0.0
        if nx == 0:
            return {"yout": (Dv * u).tolist()}
        if ns <= 1:
            return {"yout": [Dv * u[0]] if ns == 1 else []}
        dt = t[1] - t[0]
        M = np.zeros((nx+2, nx+2))
        M[:nx, :nx] = A * dt
        M[:nx, nx] = (B * dt).ravel()
        M[nx, nx+1] = 1.0
        eT = expm(M.T)
        Ad = np.ascontiguousarray(eT[:nx, :nx].T)
        Bd1 = np.ascontiguousarray(eT[nx+1, :nx].T)
        Bd0 = np.ascontiguousarray((eT[nx, :nx] - eT[nx+1, :nx]).T)
        Cf = np.ascontiguousarray(C.ravel())
        if nx == 1: yout = _sim1(Ad, Bd0, Bd1, Cf, u, Dv)
        elif nx == 2: yout = _sim2(Ad, Bd0, Bd1, Cf, u, Dv)
        elif nx == 3: yout = _sim3(Ad, Bd0, Bd1, Cf, u, Dv)
        elif nx == 4: yout = _sim4(Ad, Bd0, Bd1, Cf, u, Dv)
        elif nx == 5: yout = _sim5(Ad, Bd0, Bd1, Cf, u, Dv)
        else: yout = _sim_gen(Ad, Bd0, Bd1, Cf, u, Dv)
        return {"yout": yout.tolist()}
