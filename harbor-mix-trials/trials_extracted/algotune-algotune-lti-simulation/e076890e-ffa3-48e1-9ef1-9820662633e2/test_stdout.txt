import numpy as np
from scipy import signal

class Solver:
    def solve(self, problem, **kwargs):
        """
        Solves the LTI simulation problem using scipy.signal.lsim with TransferFunction.
        TransferFunction is slightly faster than LTI for this use case.
        """
        num = problem["num"]
        den = problem["den"]
        u = problem["u"]
        t = problem["t"]

        # Create the system using TransferFunction (faster than lti)
        system = signal.TransferFunction(num, den)

        # Simulate the system response
        tout, yout, xout = signal.lsim(system, u, t)

        return {"yout": yout.tolist()}
