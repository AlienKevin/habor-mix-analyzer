from __future__ import annotations

from typing import Any

import numpy as np
from scipy.linalg import expm

try:
    from numba import njit
except Exception:  # pragma: no cover
    njit = None


if njit is not None:
    @njit(cache=False, fastmath=False)
    def _simulate_order1_numba(Ad, bd0, bd1, C, D, U):
        n_steps = U.shape[0]
        y = np.empty(n_steps, dtype=np.float64)
        x = 0.0
        y[0] = D * U[0]
        for k in range(1, n_steps):
            uk0 = U[k - 1]
            uk1 = U[k]
            x = x * Ad + bd0 * uk0 + bd1 * uk1
            y[k] = C * x + D * uk1
        return y

    @njit(cache=False, fastmath=False)
    def _simulate_order2_numba(Ad, bd0, bd1, C, D, U):
        n_steps = U.shape[0]
        y = np.empty(n_steps, dtype=np.float64)
        x0 = 0.0
        x1 = 0.0
        y[0] = D * U[0]
        for k in range(1, n_steps):
            uk0 = U[k - 1]
            uk1 = U[k]
            p0 = x0
            p1 = x1
            x0 = p0 * Ad[0, 0] + p1 * Ad[1, 0] + bd0[0] * uk0 + bd1[0] * uk1
            x1 = p0 * Ad[0, 1] + p1 * Ad[1, 1] + bd0[1] * uk0 + bd1[1] * uk1
            y[k] = C[0] * x0 + C[1] * x1 + D * uk1
        return y

    @njit(cache=False, fastmath=False)
    def _simulate_order3_numba(Ad, bd0, bd1, C, D, U):
        n_steps = U.shape[0]
        y = np.empty(n_steps, dtype=np.float64)
        x0 = 0.0
        x1 = 0.0
        x2 = 0.0
        y[0] = D * U[0]
        for k in range(1, n_steps):
            uk0 = U[k - 1]
            uk1 = U[k]
            p0 = x0
            p1 = x1
            p2 = x2
            x0 = p0 * Ad[0, 0] + p1 * Ad[1, 0] + p2 * Ad[2, 0] + bd0[0] * uk0 + bd1[0] * uk1
            x1 = p0 * Ad[0, 1] + p1 * Ad[1, 1] + p2 * Ad[2, 1] + bd0[1] * uk0 + bd1[1] * uk1
            x2 = p0 * Ad[0, 2] + p1 * Ad[1, 2] + p2 * Ad[2, 2] + bd0[2] * uk0 + bd1[2] * uk1
            y[k] = C[0] * x0 + C[1] * x1 + C[2] * x2 + D * uk1
        return y

    @njit(cache=False, fastmath=False)
    def _simulate_order4_numba(Ad, bd0, bd1, C, D, U):
        n_steps = U.shape[0]
        y = np.empty(n_steps, dtype=np.float64)
        x0 = 0.0
        x1 = 0.0
        x2 = 0.0
        x3 = 0.0
        y[0] = D * U[0]
        for k in range(1, n_steps):
            uk0 = U[k - 1]
            uk1 = U[k]
            p0 = x0
            p1 = x1
            p2 = x2
            p3 = x3
            x0 = p0 * Ad[0, 0] + p1 * Ad[1, 0] + p2 * Ad[2, 0] + p3 * Ad[3, 0] + bd0[0] * uk0 + bd1[0] * uk1
            x1 = p0 * Ad[0, 1] + p1 * Ad[1, 1] + p2 * Ad[2, 1] + p3 * Ad[3, 1] + bd0[1] * uk0 + bd1[1] * uk1
            x2 = p0 * Ad[0, 2] + p1 * Ad[1, 2] + p2 * Ad[2, 2] + p3 * Ad[3, 2] + bd0[2] * uk0 + bd1[2] * uk1
            x3 = p0 * Ad[0, 3] + p1 * Ad[1, 3] + p2 * Ad[2, 3] + p3 * Ad[3, 3] + bd0[3] * uk0 + bd1[3] * uk1
            y[k] = C[0] * x0 + C[1] * x1 + C[2] * x2 + C[3] * x3 + D * uk1
        return y

    @njit(cache=False, fastmath=False)
    def _simulate_siso_numba(Ad, bd0, bd1, C, D, U):
        n_steps = U.shape[0]
        n_states = Ad.shape[0]
        y = np.empty(n_steps, dtype=np.float64)
        x = np.zeros(n_states, dtype=np.float64)
        y[0] = D * U[0]
        prev = np.empty(n_states, dtype=np.float64)
        for k in range(1, n_steps):
            for i in range(n_states):
                prev[i] = x[i]
            uk0 = U[k - 1]
            uk1 = U[k]
            for j in range(n_states):
                s = bd0[j] * uk0 + bd1[j] * uk1
                for i in range(n_states):
                    s += prev[i] * Ad[i, j]
                x[j] = s
            out = D * uk1
            for j in range(n_states):
                out += C[j] * x[j]
            y[k] = out
        return y

    _simulate_order4_numba = None
    _simulate_siso_numba = None


def _simulate_siso_py(Ad, bd0, bd1, C, D, U):
    n_steps = U.shape[0]
    y = np.empty(n_steps, dtype=np.float64)
    x = np.zeros(Ad.shape[0], dtype=np.float64)
    y[0] = D * U[0]
    for k in range(1, n_steps):
        x = x @ Ad + bd0 * U[k - 1] + bd1 * U[k]
        y[k] = C @ x + D * U[k]
    return y


class Solver:
    def __init__(self):
        # Compile the hot loop during initialization; init time is excluded by the harness.
        if _simulate_siso_numba is not None:
            Ad = np.eye(2, dtype=np.float64)
            b0 = np.ones(2, dtype=np.float64) * 0.1
            b1 = np.ones(2, dtype=np.float64) * 0.2
            C = np.ones(2, dtype=np.float64)
            U = np.linspace(0.0, 1.0, 4, dtype=np.float64)
            _simulate_order1_numba(1.0, 0.1, 0.2, 1.0, 0.0, U)
            _simulate_order2_numba(Ad, b0, b1, C, 0.0, U)
            Ad3 = np.eye(3, dtype=np.float64)
            b3 = np.ones(3, dtype=np.float64) * 0.1
            C3 = np.ones(3, dtype=np.float64)
            _simulate_order3_numba(Ad3, b3, b3, C3, 0.0, U)
            Ad4 = np.eye(4, dtype=np.float64)
            b4 = np.ones(4, dtype=np.float64) * 0.1
            C4 = np.ones(4, dtype=np.float64)
            _simulate_order4_numba(Ad4, b4, b4, C4, 0.0, U)
            _simulate_siso_numba(Ad, b0, b1, C, 0.0, U)

    def solve(self, problem, **kwargs) -> Any:
        num = np.asarray(problem["num"], dtype=np.float64).ravel()
        den = np.asarray(problem["den"], dtype=np.float64).ravel()
        U = np.asarray(problem["u"], dtype=np.float64).ravel()
        T = np.asarray(problem["t"], dtype=np.float64).ravel()

        n_steps = T.size
        if n_steps == 0:
            return {"yout": []}

        # Match scipy.signal.normalize/tf2ss for ordinary SISO inputs.
        nz_den = np.flatnonzero(den != 0.0)
        if nz_den.size and nz_den[0] > 0:
            den = den[nz_den[0]:]
        den0 = den[0]
        if den0 != 1.0:
            den = den / den0
            num = num / den0

        nz_num = np.flatnonzero(np.abs(num) > 1e-14)
        if nz_num.size == 0:
            num = num[-1:]
        elif nz_num[0] > 0:
            num = num[nz_num[0]:]

        K = den.size
        M = num.size
        if M < K:
            padded = np.zeros(K, dtype=np.float64)
            padded[K - M:] = num
            num = padded
        elif M > K:
            # Let SciPy raise the same kind of exception for invalid improper systems.
            from scipy import signal
            _, yout, _ = signal.lsim(signal.lti(num, den), U, T)
            return {"yout": np.asarray(yout, dtype=np.float64).tolist()}

        if K == 1:
            return {"yout": (num[0] * U).tolist()}

        if n_steps == 1:
            return {"yout": [float(num[0] * U[0])]}

        if not np.any(U):
            return {"yout": [0.0] * n_steps}

        dt = float(T[1] - T[0])
        n_states = K - 1

        # Controller canonical form used by scipy.signal.tf2ss.
        A = np.zeros((n_states, n_states), dtype=np.float64)
        A[0, :] = -den[1:]
        if n_states > 1:
            idx = np.arange(n_states - 1)
            A[idx + 1, idx] = 1.0
        B = np.zeros(n_states, dtype=np.float64)
        B[0] = 1.0
        D = float(num[0])
        C = num[1:] - D * den[1:]

        # Same augmented matrix exponential as scipy.signal.lsim(interp=True),
        # with the single-input blocks flattened for faster simulation.
        Maug = np.zeros((n_states + 2, n_states + 2), dtype=np.float64)
        Maug[:n_states, :n_states] = A * dt
        Maug[:n_states, n_states] = B * dt
        Maug[n_states, n_states + 1] = 1.0
        E = expm(Maug.T)
        Ad = np.ascontiguousarray(E[:n_states, :n_states])
        bd1 = np.ascontiguousarray(E[n_states + 1, :n_states])
        bd0 = np.ascontiguousarray(E[n_states, :n_states] - bd1)
        C = np.ascontiguousarray(C)
        U = np.ascontiguousarray(U)

        if _simulate_siso_numba is not None:
            if n_states == 1:
                yout = _simulate_order1_numba(Ad[0, 0], bd0[0], bd1[0], C[0], D, U)
            elif n_states == 2:
                yout = _simulate_order2_numba(Ad, bd0, bd1, C, D, U)
            elif n_states == 3:
                yout = _simulate_order3_numba(Ad, bd0, bd1, C, D, U)
            elif n_states == 4:
                yout = _simulate_order4_numba(Ad, bd0, bd1, C, D, U)
            else:
                yout = _simulate_siso_numba(Ad, bd0, bd1, C, D, U)
        else:
            yout = _simulate_siso_py(Ad, bd0, bd1, C, D, U)
        return {"yout": yout.tolist()}
