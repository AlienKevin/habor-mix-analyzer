import numpy as np
from scipy.signal import tf2ss
from scipy import linalg
from numba import njit
from typing import Any


@njit(cache=True)
def _step_states(xout, Ad, u, bd0, bd1, n_steps):
    """Step through states with linear interpolation input (FOH)."""
    for i in range(1, n_steps):
        for s in range(xout.shape[1]):
            val = 0.0
            for ss in range(xout.shape[1]):
                val += xout[i - 1, ss] * Ad[ss, s]
            xout[i, s] = val + u[i - 1] * bd0[s] + u[i] * bd1[s]


# Warm up numba compilation at module load time
_dummy_x = np.zeros((2, 2))

_dummy_u = np.zeros(2)
_dummy_b = np.zeros(2)
_step_states(_dummy_x, _dummy_Ad, _dummy_u, _dummy_b, _dummy_b, 2)


class Solver:
    def solve(self, problem: dict[str, Any], **kwargs) -> dict[str, list[float]]:
        """
        Solves the LTI simulation problem using optimized state-space simulation.

        Uses tf2ss for state-space conversion, matrix exponential for
        discretization with first-order hold (linear interpolation), and
        numba-accelerated time-stepping.
        """
        num = problem["num"]
        den = problem["den"]
        u = np.asarray(problem["u"], dtype=float)
        t = np.asarray(problem["t"], dtype=float)

        n_steps = len(t)

        # Handle single time point
        if n_steps == 1:
            # With zero initial conditions, y = D @ u
            _, _, _, D = tf2ss(num, den)
            D = np.asarray(D, dtype=float)
            y0 = (u[0] * D.flatten()[0]).item()
            return {"yout": [y0]}

        dt = t[1] - t[0]

        # Convert transfer function to state-space
        A, B, C, D = tf2ss(num, den)
        A = np.asarray(A, dtype=float)
        B = np.asarray(B, dtype=float)
        C = np.asarray(C, dtype=float)
        D = np.asarray(D, dtype=float)

        n_states = A.shape[0]
        n_inputs = B.shape[1]

        # Handle pure gain (no dynamics)
        if n_states == 0:
            yout = u * D.flatten()[0]
            return {"yout": yout.tolist()}

        # Build augmented matrix for linear interpolation (FOH)
        # Same formulation as scipy.signal.lsim with interp=True
        M = np.zeros((n_states + 2 * n_inputs, n_states + 2 * n_inputs))
        M[:n_states, :n_states] = A * dt
        M[:n_states, n_states:n_states + n_inputs] = B * dt
        M[n_states:n_states + n_inputs, n_states + n_inputs:] = np.eye(n_inputs)

        # Matrix exponential of augmented system
        expMT = linalg.expm(M.T)
        Ad = np.ascontiguousarray(expMT[:n_states, :n_states])
        Bd1 = np.ascontiguousarray(expMT[n_states + n_inputs:, :n_states])
        Bd0 = np.ascontiguousarray(
            expMT[n_states:n_states + n_inputs, :n_states] - Bd1
        )

        # Flatten for numba
        bd0 = Bd0.flatten()
        bd1 = Bd1.flatten()

        # Time-stepping with numba acceleration
        xout = np.zeros((n_steps, n_states))
        _step_states(xout, Ad, u, bd0, bd1, n_steps)

        # Compute output
        yout = xout @ C.flatten() + u * D.flatten()[0]

        return {"yout": yout.tolist()}
